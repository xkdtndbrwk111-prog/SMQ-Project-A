              Session.getEffectiveUser().getEmail() || ''
                ).toLowerCase();

                  if (email !== C3_TEST_ACCOUNT_) {
                      throw new Error('C3_STOP_ACCOUNT');
                        }

                          if (ScriptApp.getScriptId() !== C3_TEST_SCRIPT_ID_) {
                              throw new Error('C3_STOP_SCRIPT');
                                }

                                  const ss = SpreadsheetApp.getActiveSpreadsheet();

                                    if (
                                        !ss ||
                                            ss.getId() !== C3_TEST_SPREADSHEET_ID_ ||
                                                ss.getName() !== C3_TEST_SPREADSHEET_NAME_
                                                  ) {
                                                      throw new Error('C3_STOP_RESOURCE');
                                                        }

                                                          if (ScriptApp.getProjectTriggers().length !== 0) {
                                                              throw new Error('C3_STOP_TRIGGER_PRESENT');
                                                                }

                                                                  return ss;
                                                                  }

/* C3-REV2 — Prepare fresh synthetic test date
   TEST ONLY: 2026-10-12
   Append-only. Never reset existing evidence.
*/

function c3PrepareRev2Date() {
  const ss = c3Guard_();
  const dateKey = '2026-10-12';

  const storageId =
    PropertiesService.getScriptProperties()
      .getProperty(SMQ_USAGE_SPREADSHEET_ID_KEY_U1);

  if (storageId !== ss.getId()) {
    throw new Error('C3_REV2_STORAGE_MISMATCH');
  }

  const eventSh = ss.getSheetByName('EVENT_LOG');
  const manifestSh = ss.getSheetByName(
    SMQ_USAGE_ROLLUP_MANIFEST_SHEET_U3S
  );
  const totalsSh = ss.getSheetByName(
    SMQ_USAGE_ROLLUP_TOTALS_SHEET_U3S
  );
  const dimsSh = ss.getSheetByName(
    SMQ_USAGE_ROLLUP_DIMENSIONS_SHEET_U3S
  );

  if (!eventSh || !manifestSh || !totalsSh || !dimsSh) {
    throw new Error('C3_REV2_SHEET_MISSING');
  }

  function dateExists(sh) {
    const last = sh.getLastRow();
    if (last < 2) return false;

    return sh.getRange(2, 1, last - 1, 1)
      .getValues()
      .some(row =>
        smqUsageStoredDateKeyU3S_(row[0]) === dateKey
      );
  }

  const lock = LockService.getScriptLock();

  if (!lock.tryLock(10000)) {
    throw new Error('C3_REV2_LOCK_TIMEOUT');
  }

  try {
    // Existing four synthetic events must be preserved.
    if (eventSh.getLastRow() !== 5) {
      throw new Error('C3_REV2_RAW_BASELINE_MISMATCH');
    }

    const headers = eventSh.getRange(1, 1, 1, 15)
      .getDisplayValues()[0];

    if (
      headers[0] !== 'eventId' ||
      headers[2] !== 'date' ||
      headers[14] !== 'titles'
    ) {
      throw new Error('C3_REV2_HEADERS_MISMATCH');
    }

    if (
      smqUsageRecoveryRowsU3S_(manifestSh, dateKey)
        .length !== 0 ||
      dateExists(totalsSh) ||
      dateExists(dimsSh)
    ) {
      throw new Error('C3_REV2_DATE_ALREADY_USED');
    }

    const existingDates = eventSh
      .getRange(2, 3, 4, 1)
      .getValues();

    if (existingDates.some(row =>
      smqUsageStoredDateKeyU3S_(row[0]) === dateKey
    )) {
      throw new Error('C3_REV2_RAW_DATE_ALREADY_USED');
    }

    const rows = [0, 1].map(i => [
      'C3_REV2_DUP_20261012',
      new Date('2026-10-12T01:0' + i + ':00Z'),
      dateKey,
      i === 0 ? 'SOLO' : 'TEAM',
      'PUBLIC',
      '5',
      'SYNTHETIC',
      'SYNTHETIC',
      'A',
      1,
      i === 0 ? 1 : 2,
      '123',
      'PUBLIC:synthetic',
      'U3S1',
      '["A"]'
    ]);

    eventSh.getRange(
      eventSh.getLastRow() + 1,
      1,
      rows.length,
      15
    ).setValues(rows);

    SpreadsheetApp.flush();

  } finally {
    lock.releaseLock();
  }

  // Intentionally produce a V1 duplicate-ID failure.
  let expectedError = '';

  try {
    runUsageRollupDateU3S1_(dateKey, false);
  } catch (e) {
    expectedError = String(e && e.message || e);
  }

  const records = smqUsageRecoveryRowsU3S_(
    manifestSh, dateKey
  );

  if (
    !expectedError.startsWith(
      'USAGE_ROLLUP_DUPLICATE_EVENT_ID'
    ) ||
    records.length !== 1 ||
    records[0].seq !== 1 ||
    String(records[0].r[3]) !==
      SMQ_USAGE_ROLLUP_SCHEMA_U3S ||
    String(records[0].r[4]) !== 'FAILED' ||
    eventSh.getLastRow() !== 7
  ) {
    throw new Error(
      'C3_REV2_PREPARATION_FAILED:' +
      expectedError.slice(0, 100)
    );
  }

  const result = {
    ok: true,
    date: dateKey,
    rawRows: eventSh.getLastRow() - 1,
    generation: String(records[0].r[2]),
    status: String(records[0].r[4]),
    expectedError: 'USAGE_ROLLUP_DUPLICATE_EVENT_ID',
    triggers: ScriptApp.getProjectTriggers().length
  };

  console.log(
    'C3_REV2_PREPARE_PASS=' + JSON.stringify(result)
  );

  return result;
}
